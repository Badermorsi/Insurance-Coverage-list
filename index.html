/* ========================================================= QUANTITY CALCULATOR SEARCH + VLOOKUP FROM THIQA COVERAGE LIST ========================================================= */ /* Find a column name safely. This makes the calculator work even if the JSON uses slightly different capitalization. */ function findThiqaColumn(possibleNames){ const availableColumns = columns.thiqa || []; for(const wanted of possibleNames){ const exact = availableColumns.find( column => String(column).toLowerCase() === String(wanted).toLowerCase() ); if(exact){ return exact; } } /* Fallback: search by partial column name. */ for(const wanted of possibleNames){ const partial = availableColumns.find( column => String(column) .toLowerCase() .includes( String(wanted).toLowerCase() ) ); if(partial){ return partial; } } return null; } /* ========================================================= CLOSE QUANTITY DROPDOWN ========================================================= */ function closeQuantityDropdown(event){ const searchWrap = document.querySelector( ".quantity-search-wrap" ); const dropdown = document.getElementById( "quantityDropdown" ); if( !searchWrap || !dropdown ){ return; } /* Do not close when clicking inside the search box or dropdown. */ if( !searchWrap.contains( event.target ) ){ dropdown.classList.remove( "show" ); } } /* ========================================================= SEARCH QUANTITY ITEMS ========================================================= */ function searchQuantityItems(){ const input = document.getElementById( "quantitySearch" ); const dropdown = document.getElementById( "quantityDropdown" ); if( !input || !dropdown ){ return; } /* Make sure Thiqa data has been loaded. */ if( !Array.isArray(thiqaData) || thiqaData.length === 0 ){ dropdown.innerHTML = '
' + 'Thiqa Coverage List is still loading...' + '
'; dropdown.classList.add( "show" ); return; } const searchText = input.value .trim() .toLowerCase(); /* If nothing has been typed, show nothing. */ if(searchText === ""){ dropdown.innerHTML = ""; dropdown.classList.remove( "show" ); return; } /* Find the actual column names from Thiqa JSON. */ const proactiveNameColumn = findThiqaColumn([ "proactive name", "Proactive Name" ]); const proactiveIdColumn = findThiqaColumn([ "proactive id", "Proactive ID", "proactive code", "Proactive Code", "id", "code" ]); if( !proactiveNameColumn && !proactiveIdColumn ){ dropdown.innerHTML = '
' + 'Proactive Name / Proactive ID columns were not found.' + '
'; dropdown.classList.add( "show" ); return; } /* PREFIX SEARCH Example: "bufe" will match: Bufecort Bufederm Bufex ... It will NOT match a word appearing in the middle of the name. */ const matches = thiqaData.filter( row => { const name = proactiveNameColumn ? String( row[ proactiveNameColumn ] ?? "" ) .trim() .toLowerCase() : ""; const id = proactiveIdColumn ? String( row[ proactiveIdColumn ] ?? "" ) .trim() .toLowerCase() : ""; return ( name.startsWith( searchText ) || id.startsWith( searchText ) ); } ); /* Limit the dropdown to avoid rendering hundreds/thousands of options. */ const results = matches.slice( 0, 50 ); dropdown.innerHTML = ""; if(results.length === 0){ dropdown.innerHTML = '
' + 'No matching Proactive Name or Proactive ID.' + '
'; dropdown.classList.add( "show" ); return; } /* Create dropdown options. */ const fragment = document.createDocumentFragment(); results.forEach( row => { const option = document.createElement( "div" ); option.className = "quantity-option"; const id = proactiveIdColumn ? "" ) : ""; /* Show: ID — FULL PROACTIVE NAME */ option.innerHTML = '' + escapeQuantityHtml(id) + '' + '' + escapeQuantityHtml(name) + ''; /* Save the actual Thiqa row. This is the VLOOKUP result. */ option.addEventListener( "click", function(event){ event.stopPropagation(); selectQuantityItem( row ); } ); fragment.appendChild( option ); } ); dropdown.appendChild( fragment ); dropdown.classList.add( "show" ); } /* ========================================================= SELECT QUANTITY ITEM ========================================================= */ function selectQuantityItem(row){ const dropdown = document.getElementById( "quantityDropdown" ); const search = document.getElementById( "quantitySearch" ); /* Find the five required columns. */ const proactiveIdColumn = findThiqaColumn([ "proactive id", "Proactive ID", "proactive code", "Proactive Code", "id", "code" ]); const proactiveNameColumn = findThiqaColumn([ "proactive name", "Proactive Name" ]); const dosageFormColumn = findThiqaColumn([ "dosage form", "Dosage Form" ]); const strengthColumn = findThiqaColumn([ "strength", "Strength" ]); const packageSizeColumn = findThiqaColumn([ "package size", "Package Size" ]); /* VLOOKUP RESULTS */ const proactiveId = proactiveIdColumn ? row[ proactiveIdColumn ] ?? "" : ""; const proactiveName = proactiveNameColumn ? row[ proactiveNameColumn ] ?? "" : ""; const dosageForm = dosageFormColumn ? row[ dosageFormColumn ] ?? "" : ""; const strength = strengthColumn ? row[ strengthColumn ] ?? "" : ""; const packageSize = packageSizeColumn ? row[ packageSizeColumn ] ??
